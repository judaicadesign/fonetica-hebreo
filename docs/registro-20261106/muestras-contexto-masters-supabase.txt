וְאָכַלְתָּ וְשָׂבָעְתָּ׃ הִשָּׁמְרוּ
וְאָכַלְתָּ וְשָׂבָֽעְתָּ, וּבֵרַכְתָּ אֶת־יְיָ
וְאָהַבְתָּ אֵת יְיָ אֱלֹהֶיךָ בְּכָל־לְבָבְךָ וּבְכָל־נַפְשְׁךָ וּבְכָל־מְאֹדֶךָ׃ וְהָיוּ
וְשִׁנַּנְתָּם לְבָנֶיךָ וְדִבַּרְתָּ בָּם, בְּשִׁבְתְּךָ בְּבֵיתֶךָ
וְנָתַתִּי מְטַר אַרְצְכֶם בְּעִתּוֹ יוֹרֶה וּמַלְקוֹשׁ וְאָסַפְתָּ דְגָנֶךָ
וְנָתַתִּי עֵשֶׂב בְּשָׂדְךָ לִבְהֶמְתֶּךָ
וְאָמַרְתָּ אֲלֵהֶם, וְשָׁמַרְתָּ כָּל־חֻקָּיו
וְהַאֲזַנְתָּ לְמִצְו‍ֹתָיו וְשָׁמַרְתָּ כָּל־חֻקָּיו כָּל־הַמַּחֲלָה אֲשֶׁר
אִם תִּשְׁכַּב לֹא תִפְחָד וְשָׁכַבְתָּ וְעָרְבָה שְׁנָתֶךָ
וְאַתָּה בְּרַחֲמֶיךָ הָרַבִּים הֵפַרְתָּ אֶת עֲצָתוֹ וְקִלְקַלְתָּ אֶת מַחֲשַׁבְתּוֹ
עָמַדְתָּ לָהֶם בְּעֵת צָרָתָם רַבְתָּ אֶת רִיבָם וּלְךָ עָשִׂיתָ שֵׁם גָּדוֹל וְעָשִׂיתָ עִמָּהֶם נֵס
עַל שֶׁרָצִיתָ וְהִנְחַֽלְתָּ לַאֲבוֹתֵֽינוּ
בִּקְדֻשָּׁה וּבְטָהֳרָה, וְטָהֳרָה מִשִּׁבְעָה
מִקֶּטֶב יָשׁוּד צָהֳרָיִם. יִפֹּל מִצִּדְּךָ
חָרְבַּן בֵּיתְךָ מָרְדְּכַי וְאֶסְתֵּר בְּעָוֹן חוֹלָלְתִּי וּבְחֵטְא
אֵלִי אַתָּה וְאוֹדֶךָּ אֱלֹהַי נוֹשָֽׁעְתִּי.
כָּל־חָי אַתָּה מִכָּל־רָע יִשְׁמֹר וְכָל־צְבָאָם׃ וַיְכַל
כִּי בָנוּ בָחַרְתָּ וְאוֹתָנוּ קִדַּשְׁתָּ מִכָּל־הָעַמִּים
